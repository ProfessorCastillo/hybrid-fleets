	vHofQ = 0.0;
	vHofQ += (costCSDriverArrayList.get(0)-0)*F_of_x(middle+0);
	for (int i = 0; i < n-1; i++) {
		vHofQ += (costCSDriverArrayList.get(i+1)-costCSDriverArrayList.get(i)) * F_of_x(middle+numCSTypeCumulArrayList.get(i));
	}
	vHofQ += (pP-costCSDriverArrayList.get(n-1))*F_of_x(middle+numCSTypeCumulArrayList.get(n-1));
	vQ = middle;
}


traceln("H(Q): "+vHofQ+" | "+"goal, R: "+vR);
traceln("found Q*: "+vQ);
